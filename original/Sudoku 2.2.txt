procedure Borda;  { procedure p/ desenhar a borda }

     for a:= y1 to y2 do

                  gotoxy(b, a);

////////////////////////////////////////////////////////////////////////////////
procedure Datilografar(text: string);  { efeito: digita cada caractere do texto por vez }

var
            c:              byte;
            tempo:          byte;
            wait:           boolean;
            aux:            char;

begin
     tempo:= 50;
     wait:= false;
     for c:= 1 to length(text) do
         begin
              write(text[c]);
              delay(tempo);
              if keypressed then
                 begin
                      tempo:= 0;
                      wait:= true;
                 end;
         end;
     if wait then
        aux:= readkey;
        if aux= #0 then
           readkey;
     aux:= readkey;
     if aux= #0 then
        readkey;
end;  { fim procedure Datilografar }

procedure GradeMolde;  { desenha o esqueleto da grade, sem numeros }

          a, b:         word;

begin
     textcolor(2);
     write(#218);  { primeira linha }
     for b:= 2 to 18 do

            write(#196)
         else
             write(#194);
     writeln(#191);
     for a:= 2 to 18 do  { linhas do meio }
         if a mod 2= 0 then
            begin
                 for b:= 1 to 9 do
                     write(#179, ' ');

            end

                  write(#195);
                  for b:= 2 to 18 do
                      if b mod 2= 0 then

                      else

             end;
     write(#192);  { ultima linha }
     for b:= 2 to 18 do
         if b mod 2= 0 then
            write(#196)
         else

procedure WritenaGrade(what: char; erro: boolean);  { especifica p/ escrever numeros na grade, varia cor do texto e fundo }

var
          coruser, corfixo:        byte;

begin
     x:= wherex;
     y:= wherey;
     corfixo:= 10;  { cor dos numeros fixos sempre e verde claro }
     if erro then  { se for um numero errado digitado pelo usuario }
        begin
             coruser:= 12;  { muda a cor do numero p/ vermelho }
        end
     else  { se for um numero possivel }
         begin
              coruser:= 15;  { muda a cor do numero p/ branco }
         end;

     if Binstr[x div 2 + (y div 2 - 1)*9] = '0' then
        if GDB[optionnivel][optiongrade][2][x div 2 + (y div 2 - 1)*9]= '0' then
           begin
                textcolor(coruser);
                if ((x div 7) + (y div 7)) mod 2 = 0 then
                   textbackground(1);
           end
        else
            begin
                 textcolor(corfixo);
                 if ((x div 7) + (y div 7)) mod 2 = 0 then
                    textbackground(1);
            end;

     write(what);

     textbackground(0);
     textcolor(cortexto);
     gotoxy(x, y);
end;  { fim procedure WritenaGrade }

procedure PaintBack(t: byte);

var
          a, b:          byte;
          count:          byte;

begin
     count:= 0;
     for a:= 1 to 9 do
         for b:= 1 to 9 do
             begin
                  count:= count+1;  { aki preciso de três contadores: dois para as coordenadas (a e b) e um para a posicao na string (cont) }
                  gotoxy(b*2, a*2);

                  if GDB[optionnivel][optiongrade][2][count]= '1' then
                     begin
                          WritenaGrade(GDB[optionnivel][optiongrade][1][count], false);
                          delay(t);
                     end
                  else
                      WritenaGrade(' ', false);
             end;
end;  { fim procedure PaintBack }

procedure Conferir(tecla: char);  { confere se tem algum numero igual ao que foi digitado pelo usuario na linha, coluna e regiao menor 3x3 }

var
          igual:                       boolean;
          a, b:                        byte;

begin
     x:= wherex;
     y:= wherey;
     igual:= false;

     a:= 1;
     while (a <= 9) and (igual= false) do  { conferindo linha }
           begin
                if (tecla = Usernume[((y div 2 - 1)*9)+a]) and (a <> x div 2) then
                   igual:= true;
                inc(a, 1);
           end;  { fim conferindo linha }

     a:= 1;
     while (a <= 9) and (igual= false) do  { conferindo coluna }
           begin
                if (tecla = Usernume[((a - 1)*9)+(x div 2)]) and (a <> y div 2) then
                   igual:= true;
                inc(a, 1);
           end;  { fim conferindo coluna }

     a:= 1;
     while (a<= 9) and (igual= false) do  { conferindo quadrados menores }
           begin
                if (a*2 div 7) = (y div 7) then
                   begin
                        b:= 1;
                        while (b <= 9) and (igual= false) do
                              begin
                                   if (b*2 div 7) = (x div 7) then
                                      if (a <> y div 2) and (b <> x div 2) then
                                         if tecla = Usernume[(a - 1)*9+b] then
                                            igual:= true;
                                   inc(b, 1);
                              end;
                   end;
                inc(a, 1);
           end;  { fim conferindo quadrados menores }

     if igual= true then  { se foi encontrado algum numero igual }
        begin
             WritenaGrade(tecla, true);  { escreve o numero com a cor diferente (azul) }
             InfoWindow(3);  { escreve: "OPS! Esse numero nao pode ir ai!" }
        end
     else
         begin
              WritenaGrade(tecla, false);  { se o numero puder ser colocado }
         end;
end;  { fim procedure Conferir }
